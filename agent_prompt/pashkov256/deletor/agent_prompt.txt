type Config struct {

	exclude     string

	maxSizeStr  string

// parseSize converts a human-readable size string (e.g., "10KB", "1.5MB") to bytes.
func parseSize(sizeStr string) (int64, error) {
	if sizeStr == "" {
		return 0, nil
	}

	re := regexp.MustCompile(`(?i)^(\d+(\.\d+)?)\s*(b|kb|mb|gb|tb)?$`)
	matches := re.FindStringSubmatch(strings.ToLower(sizeStr))

	if len(matches) == 0 {
		return 0, fmt.Errorf("invalid size format: %s", sizeStr)
	}

	value, err := strconv.ParseFloat(matches[1], 64)
	if err != nil {
		return 0, err
	}

	unit := "b"
	if len(matches) > 3 && matches[3] != "" {
		unit = matches[3]
	}

	var multiplier int64
	switch unit {
	case "b":
		multiplier = 1
	case "kb":
		multiplier = 1024
	case "mb":
		multiplier = 1024 * 1024
	case "gb":
		multiplier = 1024 * 1024 * 1024
	case "tb":
		multiplier = 1024 * 1024 * 1024 * 1024
	default:
		return 0, fmt.Errorf("unknown size unit: %s", unit)
	}

	return int64(value * float64(multiplier)), nil
}

// parseDuration converts a duration string (e.g., "1d", "2h", "30m") to a time.Time.
func parseDuration(durationStr string) (time.Time, error) {
	if durationStr == "" {
		return time.Time{}, nil

	}

	re := regexp.MustCompile(`^(\d+)(s|m|h|d|w|M|y)$`)
	matches := re.FindStringSubmatch(durationStr)
	if matches == nil {
		return time.Time{}, fmt.Errorf("invalid duration format: %s", durationStr)
	}

	value, _ := strconv.Atoi(matches[1])
	unit := matches[2]
	var duration time.Duration

	switch unit {
	case "s":
		duration = time.Duration(value) * time.Second
	case "m":
		duration = time.Duration(value) * time.Minute
	case "h":
		duration = time.Duration(value) * time.Hour
	case "d":
		duration = time.Duration(value) * 24 * time.Hour
	case "w":
		duration = time.Duration(value) * 7 * 24 * time.Hour
	case "M":
		duration = time.Duration(value) * 30 * 24 * time.Hour // Approximation
	case "y":
		duration = time.Duration(value) * 365 * 24 * time.Hour // Approximation
	default:
		return time.Time{}, fmt.Errorf("unknown duration unit: %s", unit)
	}

	return time.Now().Add(-duration), nil
}

// formatBytes converts bytes to a human-readable string.
func formatBytes(b int64) string {
	const unit = 1024
	if b < unit {
		return fmt.Sprintf("%d B", b)
	}
	div, exp := int64(unit), 0
	for n := b / unit; n >= unit; n /= unit {
		div *= unit
		exp++
	}
	return fmt.Sprintf("%.2f %cB", float64(b)/float64(div), "KMGTPE"[exp])
}

func main() {
	var cfg Config

	flag.BoolVar(&cfg.cli, "cli", false, "Enable command-line interface mode")
	flag.StringVar(&cfg.dir, "d", ".", "Directory to scan")
	flag.StringVar(&cfg.extensions, "e", "", "Comma-separated list of file extensions to delete")
	flag.StringVar(&cfg.exclude, "exclude", "", "Comma-separated list of files/directories to exclude")
	flag.StringVar(&cfg.minSizeStr, "min-size", "", "Minimum file size (e.g., 10KB, 1MB)")
	flag.StringVar(&cfg.maxSizeStr, "max-size", "", "Maximum file size (e.g., 10KB, 1MB)")
	flag.StringVar(&cfg.older, "older", "", "Delete files older than this (e.g., 30d, 24h, 5m)")
	flag.StringVar(&cfg.newer, "newer", "", "Delete files newer than this (e.g., 30d, 24h, 5m)")
	flag.BoolVar(&cfg.subdirs, "subdirs", false, "Include subdirectories")
	flag.BoolVar(&cfg.pruneEmpty, "prune-empty", false, "Delete empty directories after scan")
	flag.BoolVar(&cfg.skipConfirm, "skip-confirm", false, "Skip confirmation before deleting")

	flag.Parse()

	if !cfg.cli {
		fmt.Println("This is a CLI tool. Please use the -cli flag.")
		return
	}

	var err error
	cfg.minSizeBytes, err = parseSize(cfg.minSizeStr)
	if err != nil {
		fmt.Println("Error parsing min-size:", err)
		return
	}
	cfg.maxSizeBytes, err = parseSize(cfg.maxSizeStr)
	if err != nil {
		fmt.Println("Error parsing max-size:", err)
		return
	}

	cfg.olderThan, err = parseDuration(cfg.older)
	if err != nil {
		fmt.Println("Error parsing older:", err)
		return
	}
	cfg.newerThan, err = parseDuration(cfg.newer)
	if err != nil {
		fmt.Println("Error parsing newer:", err)
		return
	}

	if cfg.exclude != "" {
		cfg.excludeList = strings.Split(cfg.exclude, ",")
	}
	if cfg.extensions != "" {
		cfg.extList = strings.Split(cfg.extensions, ",")
	}

	var filesToDelete []string
	var totalSize int64

	walkFunc := func(path string, info os.FileInfo, err error) error {
		if err != nil {
			return err
		}

		if info.IsDir() {
			// Skip excluded directories
			for _, ex := range cfg.excludeList {
				if info.Name() == ex {
					return filepath.SkipDir
				}
			}
			return nil
		}

		// File extension filter
		if len(cfg.extList) > 0 {
			match := false
			for _, ext := range cfg.extList {
				if strings.HasSuffix(strings.ToLower(info.Name()), "."+strings.ToLower(ext)) {
					match = true
					break
				}
			}
			if !match {
				return nil
			}
		}

		// File size filters
		if cfg.minSizeBytes > 0 && info.Size() < cfg.minSizeBytes {
			return nil
		}
		if cfg.maxSizeBytes > 0 && info.Size() > cfg.maxSizeBytes {
			return nil
		}

		// Time-based filters
		if !cfg.olderThan.IsZero() && info.ModTime().After(cfg.olderThan) {
			return nil

		}
		if !cfg.newerThan.IsZero() && info.ModTime().Before(cfg.newerThan) {
			return nil
		}

		// Exclude specific files
		for _, ex := range cfg.excludeList {
			if info.Name() == ex {
				return nil
			}
		}

		filesToDelete = append(filesToDelete, path)
		totalSize += info.Size()
		fmt.Printf("%-10s %s\n", formatBytes(info.Size()), path)
		return nil
	}

	if cfg.subdirs {
		filepath.Walk(cfg.dir, walkFunc)
	} else {
		entries, err := os.ReadDir(cfg.dir)
		if err != nil {
			fmt.Println("Error reading directory:", err)
			return
		}
		for _, entry := range entries {
			if !entry.IsDir() {
				info, err := entry.Info()
				if err == nil {
					walkFunc(filepath.Join(cfg.dir, entry.Name()), info, nil)
				}
			}
		}
	}

	if len(filesToDelete) == 0 {
		fmt.Println("No files to delete.")
		return
	}

	fmt.Printf("\n")
	if !cfg.skipConfirm {
		fmt.Printf("Are you sure you want to delete %d files? (y/N): ", len(filesToDelete))
		var response string
		fmt.Scanln(&response)
		if response != "y" && response != "Y" {
			fmt.Println("Operation cancelled.")
			return
		}
	}

	for _, file := range filesToDelete {
		err := os.Remove(file)
		if err != nil {
			fmt.Printf("Error deleting %s: %v\n", file, err)
		}
	}

	fmt.Printf("\u2713  Deleted: %s\n", formatBytes(totalSize))

	if cfg.pruneEmpty {
		fmt.Println("\nScanning for empty directories...")
		deletedDirs := 0
		for {
			deletedThisPass := 0
			filepath.Walk(cfg.dir, func(path string, info os.FileInfo, err error) error {
				if err != nil {
					return err
				}
				if info.IsDir() {
					entries, err := os.ReadDir(path)
					if err != nil {
						return err
					}
					if len(entries) == 0 {
						fmt.Printf("DIR  %s\n", path)
						err := os.Remove(path)
						if err == nil {
							deletedThisPass++
						}
					}
				}
				return nil
			})
			if deletedThisPass == 0 {
				break
			}
			deletedDirs += deletedThisPass
		}
		fmt.Printf("\n\u2713  Number of deleted empty folders: %d\n", deletedDirs)
	}
}
